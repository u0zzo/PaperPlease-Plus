TAKE_FAMILY_PASSPORTS
	" The Ministry of Information has confiscated your family's passports.
	@ TakeFamilyPassports

ESCAPE
	? Night/NumEscapePassports == 1
		" You can now escape to Obristan by yourself.
		@ Include ESCAPE_1
	? Night/NumEscapePassports > 1 && Night/Profit < 50
		@ Include ESCAPE_1

	? Night/NumEscapePassports == 2
		" You can now escape to Obristan with one family member.
		@ Include ESCAPE_2
	? Night/NumEscapePassports > 2 && Night/Profit >= 50 && Night/Profit < 75
		@ Include ESCAPE_2

	? Night/NumEscapePassports == 3
		" You can now escape to Obristan with two family members.
		@ Include ESCAPE_3
	? Night/NumEscapePassports > 3 && Night/Profit >= 75 && Night/Profit < 100
		@ Include ESCAPE_3

	? Night/NumEscapePassports == 4
		" You can now escape to Obristan with three family members.
		@ Include ESCAPE_4
	? Night/NumEscapePassports > 4 && Night/Profit >= 100 && Night/Profit < 125
		@ Include ESCAPE_4

	? Night/HaveNiece == 1
		? Night/NumEscapePassports == 5
			" You can now escape to Obristan with four family members.
			@ Include ESCAPE_5
		? Night/NumEscapePassports > 5 && Night/Profit >= 125 && Night/Profit < 150
			@ Include ESCAPE_5

		? Night/NumEscapePassports >= 6
			" You can now escape to Obristan with everyone.
			@ Include ESCAPE_6

	? Night/HaveNiece != 1
		? Night/NumEscapePassports >= 5
			" You can now escape to Obristan with everyone.
			@ Include ESCAPE_5

ESCAPE_1
	> ESCAPE (SELF) | -25 | NO

FINAL
	? Game/HelpedEzic == YES
		# On Ezic team, see if we killed the attackers today
		? Night/AttackResult == SOLVED
			? Game/WallDestroyed == YES
				@ End help-ezic-kill-attackers-wall-destroyed
			? Game/WallDestroyed != YES
				@ End help-ezic-kill-attackers-wall-safe				
		? Night/AttackResult != SOLVED
			@ End help-ezic-hold-fire
	? Game/HelpedEzic != YES
		? Game/WallDestroyed == YES
			@ End wall-destroyed
		? Game/WallDestroyed != YES
			? Game/EzicAgentsApproved < 1
				@ End ignore-ezic
			? Game/EzicAgentsApproved >= 1
				@ End help-few-ezic
